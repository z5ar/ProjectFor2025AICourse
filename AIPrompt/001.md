# 电影字幕人工智能生成器

下面是一个电影字幕人工智能生成器的实现方案，能够区分不同说话人物，并支持在配备商业级显卡的笔记本电脑上进行训练。

## 系统设计思路

### 核心功能
- 语音识别：将电影中的语音转换为文本
- 说话人区分：识别不同说话者并分配标签
- 字幕生成：生成带说话人标识的SRT格式字幕
- 时间同步：精确对齐字幕与视频时间轴

### 技术架构
```
视频输入 → 音频提取 → 语音识别 → 说话人区分 → 字幕生成 → 输出字幕文件
```

### 硬件要求
- NVIDIA GPU（如RTX 3060/3080或更高）
- 至少16GB RAM
- 足够存储空间的SSD

## 完整实现代码

```python
import torch
import torchaudio
from transformers import Wav2Vec2Processor, Wav2Vec2ForCTC
from pyannote.audio import Pipeline
from pyannote.audio.pipelines.utils.hook import ProgressHook
from moviepy.editor import VideoFileClip
import numpy as np
import os
import subprocess
from datetime import timedelta
import re

class SubtitleGenerator:
    def __init__(self, device='cuda' if torch.cuda.is_available() else 'cpu'):
        self.device = device
        self.asr_processor = None
        self.asr_model = None
        self.diarization_pipeline = None
        
    def load_models(self):
        """加载语音识别和说话人区分模型"""
        print("Loading speech recognition model...")
        self.asr_processor = Wav2Vec2Processor.from_pretrained("facebook/wav2vec2-large-960h-lv60-self")
        self.asr_model = Wav2Vec2ForCTC.from_pretrained("facebook/wav2vec2-large-960h-lv60-self").to(self.device)
        
        print("Loading speaker diarization model...")
        # 使用Hugging Face Token访问pyannote模型
        self.diarization_pipeline = Pipeline.from_pretrained(
            "pyannote/speaker-diarization-3.1",
            use_auth_token="YOUR_HF_TOKEN"  # 替换为你的Hugging Face Token
        ).to(torch.device(self.device))
        
    def extract_audio(self, video_path, audio_path="temp_audio.wav"):
        """从视频中提取音频"""
        print(f"Extracting audio from {video_path}...")
        video = VideoFileClip(video_path)
        video.audio.write_audiofile(audio_path, fps=16000, codec='pcm_s16le')
        return audio_path
    
    def transcribe_audio(self, audio_path):
        """语音识别：将音频转换为文本"""
        print("Transcribing audio...")
        
        # 加载音频文件
        waveform, sample_rate = torchaudio.load(audio_path)
        
        # 重采样为16kHz
        if sample_rate != 16000:
            resampler = torchaudio.transforms.Resample(sample_rate, 16000)
            waveform = resampler(waveform)
        
        # 处理音频
        input_values = self.asr_processor(
            waveform.squeeze().numpy(), 
            sampling_rate=16000, 
            return_tensors="pt"
        ).input_values.to(self.device)
        
        # 获取模型输出
        with torch.no_grad():
            logits = self.asr_model(input_values).logits
        
        # 解码输出
        predicted_ids = torch.argmax(logits, dim=-1)
        transcription = self.asr_processor.batch_decode(predicted_ids)[0]
        
        return transcription.lower()
    
    def diarize_speakers(self, audio_path):
        """区分不同说话者"""
        print("Diarizing speakers...")
        
        # 应用说话人区分管道
        diarization = self.diarization_pipeline(
            audio_path,
            num_speakers=2,  # 可以指定说话人数或自动检测
            hook=ProgressHook()
        )
        
        # 提取说话人片段
        speaker_segments = []
        for segment, _, speaker in diarization.itertracks(yield_label=True):
            speaker_segments.append({
                "start": segment.start,
                "end": segment.end,
                "speaker": speaker
            })
        
        return speaker_segments
    
    def align_transcription(self, transcription, speaker_segments, audio_duration):
        """将转录文本与说话人片段对齐"""
        print("Aligning transcription with speaker segments...")
        
        # 简单分词（实际应用中应使用更复杂的分词器）
        words = re.findall(r'\b\w+\b', transcription)
        word_duration = audio_duration / len(words)
        
        # 为每个单词分配时间戳
        word_timestamps = []
        for i, word in enumerate(words):
            start_time = i * word_duration
            end_time = (i + 1) * word_duration
            word_timestamps.append({
                "word": word,
                "start": start_time,
                "end": end_time
            })
        
        # 为每个说话人片段分配文本
        aligned_segments = []
        for segment in speaker_segments:
            segment_text = []
            for word in word_timestamps:
                if word["start"] >= segment["start"] and word["end"] <= segment["end"]:
                    segment_text.append(word["word"])
            
            if segment_text:
                aligned_segments.append({
                    "speaker": segment["speaker"],
                    "start": segment["start"],
                    "end": segment["end"],
                    "text": " ".join(segment_text)
                })
        
        return aligned_segments
    
    def generate_subtitles(self, aligned_segments, output_path="output.srt"):
        """生成SRT格式字幕"""
        print(f"Generating subtitles at {output_path}...")
        
        with open(output_path, "w", encoding="utf-8") as f:
            for i, segment in enumerate(aligned_segments, 1):
                # 格式化时间
                start_time = timedelta(seconds=segment["start"])
                end_time = timedelta(seconds=segment["end"])
                
                # 写入SRT格式
                f.write(f"{i}\n")
                f.write(f"{self.format_timedelta(start_time)} --> {self.format_timedelta(end_time)}\n")
                f.write(f"[{segment['speaker']}] {segment['text']}\n\n")
        
        return output_path
    
    def format_timedelta(self, td):
        """将timedelta格式化为SRT时间格式"""
        hours, remainder = divmod(td.seconds, 3600)
        minutes, seconds = divmod(remainder, 60)
        milliseconds = td.microseconds // 1000
        return f"{hours:02}:{minutes:02}:{seconds:02},{milliseconds:03}"
    
    def process_video(self, video_path, output_srt="output.srt"):
        """处理视频并生成字幕"""
        print(f"Processing video: {video_path}")
        
        # 步骤1: 提取音频
        audio_path = self.extract_audio(video_path)
        
        # 步骤2: 获取音频时长
        video = VideoFileClip(video_path)
        audio_duration = video.audio.duration
        video.close()
        
        # 步骤3: 区分说话人
        speaker_segments = self.diarize_speakers(audio_path)
        
        # 步骤4: 语音识别
        transcription = self.transcribe_audio(audio_path)
        
        # 步骤5: 对齐文本与说话人
        aligned_segments = self.align_transcription(transcription, speaker_segments, audio_duration)
        
        # 步骤6: 生成字幕
        srt_path = self.generate_subtitles(aligned_segments, output_srt)
        
        # 清理临时文件
        os.remove(audio_path)
        
        print(f"Subtitle generation complete: {srt_path}")
        return srt_path

if __name__ == "__main__":
    # 初始化字幕生成器
    generator = SubtitleGenerator()
    
    # 加载模型（首次运行会下载模型）
    generator.load_models()
    
    # 处理视频文件
    video_file = "sample_movie.mp4"  # 替换为你的视频文件
    subtitle_file = "movie_subtitles.srt"
    
    generator.process_video(video_file, subtitle_file)
    
    print("\nSubtitle preview:")
    with open(subtitle_file, "r") as f:
        print(f.read()[:500])  # 打印前500个字符预览
```

    output_dir='./results',

## 注意事项

1. 首次运行需要下载模型（约1.5GB），请确保网络连接稳定
2. 处理1小时视频大约需要15-30分钟（取决于GPU性能）
3. 说话人区分精度受音频质量影响较大
4. 对于嘈杂环境或多说话人重叠场景，效果可能下降
5. 商业使用需遵守模型许可协议（特别是pyannote模型）
